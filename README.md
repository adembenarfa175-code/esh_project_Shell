## 📦 Install via APT Repo:
\`\`\`bash
echo "deb [trusted=yes] https://raw.githubusercontent.com/adembenarfa175-code/esh_project_Shell/main/esh-repo stable main" | tee /etc/apt/sources.list.d/esh.list
apt update && apt install esh
\`\`\`
